-- Doctor slots table
CREATE TABLE doctor_slots (
    id INT AUTO_INCREMENT PRIMARY KEY,
    doctor_id INT NOT NULL,
    hospital_id INT NOT NULL,
    slot_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    slot_duration INT DEFAULT 30, -- in minutes
    max_patients INT DEFAULT 1,
    booked_count INT DEFAULT 0,
    is_active BOOLEAN DEFAULT true,
    slot_type ENUM('regular', 'walk_in') DEFAULT 'regular',
    walk_in_capacity INT DEFAULT 5, -- Max walk-ins for walk-in slots
    walk_in_booked INT DEFAULT 0, -- Current walk-ins booked
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (doctor_id) REFERENCES users(id),
    FOREIGN KEY (hospital_id) REFERENCES hospitals(id),
    UNIQUE KEY unique_slot (doctor_id, slot_date, start_time)
);

-- Indexes for performance
CREATE INDEX idx_slot_doctor_date ON doctor_slots(doctor_id, slot_date);
CREATE INDEX idx_slot_availability ON doctor_slots(slot_date, start_time, is_active, slot_type);
CREATE INDEX idx_slot_type ON doctor_slots(slot_type, is_active);
